// ============================================================
// ejemplo_semantico.txt — Programa VÁLIDO para análisis semántico
// Todos los tipos son compatibles y las variables están declaradas.
// ============================================================

// --- Tipos básicos bien asignados ---
int    entero   = 42;
float  decimal  = 3.14;
double preciso  = 2.718281828;
bool   activo   = true;
string nombre   = "Ana";
char   inicial  = 'A';

// --- Ampliación numérica válida: int → float → double ---
float  f = 10;       // int asignado a float: OK
double d = 3.14;     // double literal a double: OK

// --- Asignaciones con expresiones ---
entero = entero + 1;
decimal = decimal * 2.0;

// --- Operaciones aritméticas válidas (numérico + numérico) ---
int suma       = entero + 5;
double mezcla  = entero + preciso;

// --- Condiciones booleanas en if (deben ser bool) ---
if (entero > 10) {
    cout << entero;
}

if (activo) {
    cout << nombre;
} else {
    cout << inicial;
}

// --- Operadores lógicos (bool && bool) ---
bool rango = entero > 0 && entero < 100;
if (rango && activo) {
    cout << suma;
}

// --- Scopes: variable local solo visible dentro del bloque ---
if (entero > 0) {
    int local = entero * 2;
    cout << local;
}
// Aquí 'local' ya no existe

// --- While con condición booleana ---
int contador = 0;
while (contador < 10) {
    contador = contador + 1;
}

// --- Expresión negativa válida ---
int negativo = -entero;

// --- NOT lógico ---
bool inactivo = !activo;


// --- Funciones ---

// Función con parámetros y return
int sumar(int a, int b) {
    int c = a + b;
    return c;
}

// Función void: no retorna valor
void imprimir(string msg) {
    cout << msg;
    return;
}

// Función sin parámetros
double getPI() {
    return 3.14159;
}

// Ampliación numérica válida en return: int → double
double doble(int x) {
    return x;
}
