// ============================================================
// ejemplo_semantico_error.txt — Programa con ERRORES SEMÁNTICOS
//
// A diferencia de los errores sintácticos (estructura incorrecta),
// estos errores son de SIGNIFICADO: la estructura es válida pero
// el programa no tiene sentido semántico.
//
// El análisis semántico ACUMULA todos los errores sin detenerse.
// ============================================================

// --- ERROR 1: Variable usada sin declarar ---
// La gramática lo acepta (sintácticamente válido),
// pero semánticamente 'z' no existe.
// Mensaje: variable 'z' no fue declarada.
int resultado = z + 1;


// --- ERROR 2: Redeclaración en el mismo scope ---
// No se puede declarar 'x' dos veces en el mismo ámbito.
// Mensaje: variable 'x' ya fue declarada en este ámbito.
int x = 10;
int x = 20;


// --- ERROR 3: Incompatibilidad de tipos en declaración ---
// 'activo' es bool, no se le puede asignar un entero.
// Mensaje: no se puede asignar tipo 'int' a variable de tipo 'bool'.
bool activo = 42;


// --- ERROR 4: Incompatibilidad de tipos en asignación ---
// 'nombre' es string, no se le puede asignar un int.
// Mensaje: no se puede asignar tipo 'int' a variable de tipo 'string'.
string nombre = "Juan";
nombre = 99;


// --- ERROR 5: Operación aritmética con tipo inválido ---
// No se puede sumar string + int.
// Mensaje: el operador '+' no puede aplicarse a tipos 'string' e 'int'.
string texto = "hola";
int suma = texto + 5;


// --- ERROR 6: Condición del if no es bool ---
// El if necesita una condición booleana, no un entero.
// Mensaje: la condición del 'if' debe ser bool, pero es 'int'.
int valor = 10;
if (valor) {
    cout << valor;
}


// --- ERROR 7: Condición del while no es bool ---
// El while necesita una condición booleana.
// Mensaje: la condición del 'while' debe ser bool, pero es 'string'.
string s = "hola";
while (s) {
    cout << s;
}


// --- ERROR 8: Operador lógico con tipos no booleanos ---
// && requiere dos operandos bool, no int.
// Mensaje: el operador '&&' no puede aplicarse a tipos 'int' e 'int'.
int a = 5;
int b = 3;
bool cond = a && b;


// --- ERROR 9: Variable sin inicializar usada en expresión ---
// 'sinValor' fue declarada pero nunca se le asignó un valor.
// Mensaje: variable 'sinValor' podría no estar inicializada.
int sinValor;
int uso = sinValor + 1;


// --- ERROR 10: NOT sobre tipo no booleano ---
// ! solo funciona sobre bool.
// Mensaje: el operador '!' no puede aplicarse al tipo 'int'.
int num = 5;
bool negado = !num;


// --- ERROR 11: return fuera de una función ---
// Mensaje: 'return' usado fuera de una función.
return 42;


// --- ERROR 12: tipo de retorno incompatible ---
// Función declara int pero retorna string.
// Mensaje: tipo de retorno 'string' no es compatible con 'int'.
int getDato() {
    return "hola";
}


// --- ERROR 13: función void retorna valor ---
// Mensaje: función 'void' no puede retornar un valor.
void nada() {
    return 1;
}


// --- ERROR 14: función no-void sin valor de retorno ---
// Mensaje: función de tipo 'int' debe retornar un valor.
int sinRetorno() {
    return;
}
